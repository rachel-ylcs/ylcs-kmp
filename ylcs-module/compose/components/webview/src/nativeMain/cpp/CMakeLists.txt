cmake_minimum_required(VERSION 3.24)
project(rachel_webview LANGUAGES C CXX)

if(APPLE)
    enable_language(OBJC)
    add_library(rachel_webview STATIC macos_webview.m)
    target_include_directories(rachel_webview PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
    target_compile_options(rachel_webview PRIVATE -fobjc-arc)
    target_link_libraries(rachel_webview PUBLIC "-framework Foundation" "-framework WebKit")
    return()
elseif(NOT WIN32)
    add_library(rachel_webview STATIC webview_stub.c)
    return()
endif()

include(webview_sdk.cmake)

add_library(rachel_webview STATIC windows_webview.cpp ${WEBVIEW2_OBJECTS})
target_include_directories(rachel_webview PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(rachel_webview SYSTEM PRIVATE "${WEBVIEW2_SDK}/build/native/include")
target_compile_features(rachel_webview PRIVATE cxx_std_17)
target_compile_definitions(rachel_webview PRIVATE UNICODE _UNICODE NOMINMAX WINVER=0x0A00 _WIN32_WINNT=0x0A00)
target_compile_options(rachel_webview PRIVATE -fms-extensions)
target_compile_options(rachel_webview PRIVATE "SHELL:-idirafter \"${WINDOWS_SDK_INCLUDE}/winrt\"")
