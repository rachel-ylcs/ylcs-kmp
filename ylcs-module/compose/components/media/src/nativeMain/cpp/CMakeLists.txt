cmake_minimum_required(VERSION 3.20)
project(rachel_audio LANGUAGES C CXX)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)

if(WIN32)
    set(AUDIO_BACKEND windows_media.cpp windows_video.cpp)
elseif(APPLE)
    enable_language(OBJCXX)
    set(AUDIO_BACKEND macos_video.mm macos_video_observation.mm)
else()
    set(AUDIO_BACKEND linux_video.cpp linux_gstreamer.cpp)
endif()

add_library(rachel_audio_objects OBJECT rachel_audio.cpp miniaudio_audio.cpp miniaudio/miniaudio.c ${AUDIO_BACKEND})
target_compile_features(rachel_audio_objects PRIVATE c_std_11 cxx_std_17)
target_include_directories(rachel_audio_objects PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_include_directories(rachel_audio_objects PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/miniaudio")
target_compile_definitions(rachel_audio_objects PRIVATE MA_NO_ENCODING MA_NO_GENERATION MA_NO_NULL MA_ENABLE_ONLY_SPECIFIC_BACKENDS)
add_library(rachel_audio STATIC $<TARGET_OBJECTS:rachel_audio_objects>)
target_include_directories(rachel_audio PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(rachel_audio PUBLIC $<TARGET_PROPERTY:rachel_audio_objects,INTERFACE_LINK_LIBRARIES>)

if(WIN32)
    target_compile_features(rachel_audio_objects PRIVATE cxx_std_20)
    include(windows_winrt.cmake)
    if(MINGW)
        target_compile_definitions(rachel_audio_objects PRIVATE RACHEL_VIDEO_GPU)
    endif()
    target_compile_definitions(rachel_audio_objects PRIVATE WINVER=0x0A00 _WIN32_WINNT=0x0A00 UNICODE _UNICODE NOMINMAX MA_ENABLE_WASAPI)
    target_link_libraries(rachel_audio_objects PUBLIC runtimeobject shlwapi ole32 oleaut32 uuid windowsapp d3d11 dxgi)
elseif(APPLE)
    set_target_properties(rachel_audio_objects PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    set_source_files_properties(macos_video.mm macos_video_observation.mm PROPERTIES COMPILE_FLAGS "-fobjc-arc")
    target_compile_definitions(rachel_audio_objects PRIVATE MA_ENABLE_COREAUDIO)
    target_link_libraries(rachel_audio_objects PUBLIC "-framework AVFoundation" "-framework CoreMedia" "-framework CoreVideo" "-framework Foundation" "-framework CoreFoundation" "-framework CoreAudio" "-framework AudioToolbox" "-framework AudioUnit")
else()
    target_compile_definitions(rachel_audio_objects PRIVATE MA_ENABLE_ALSA MA_ENABLE_PULSEAUDIO)
    if(NOT CMAKE_CROSSCOMPILING)
        find_package(PkgConfig QUIET)
        if(PkgConfig_FOUND)
            pkg_check_modules(GSTREAMER QUIET gstreamer-app-1.0>=1.18 gstreamer-video-1.0>=1.18)
        endif()
    endif()
    if(GSTREAMER_FOUND)
        target_include_directories(rachel_audio_objects SYSTEM PRIVATE ${GSTREAMER_INCLUDE_DIRS})
        target_compile_options(rachel_audio_objects PRIVATE ${GSTREAMER_CFLAGS_OTHER})
    else()
        include(linux_gstreamer.cmake)
    endif()
    find_package(Threads REQUIRED)
    target_link_libraries(rachel_audio_objects PUBLIC Threads::Threads ${CMAKE_DL_LIBS} m)
endif()
