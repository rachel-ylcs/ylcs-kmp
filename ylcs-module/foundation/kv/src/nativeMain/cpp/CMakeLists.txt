cmake_minimum_required(VERSION 3.20)
project(rachel_kv LANGUAGES C CXX)

set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(zlib "" CACHE FILEPATH "" FORCE)
set(MMKV_CORE_SOURCE "${CMAKE_CURRENT_SOURCE_DIR}/mmkv/Core")
if(RACHEL_NATIVE_TARGET)
    # Kotlin/Native ships an older libstdc++; adapt the build copy without changing the submodule.
    set(MMKV_CORE_SOURCE "${CMAKE_CURRENT_BINARY_DIR}/mmkv-core-source")
    file(COPY "${CMAKE_CURRENT_SOURCE_DIR}/mmkv/Core/" DESTINATION "${MMKV_CORE_SOURCE}")
    file(READ "${MMKV_CORE_SOURCE}/MMKV.h" MMKV_HEADER)
    string(REPLACE "using MMKVKey_t = std::string_view;" "using MMKVKey_t = const std::string &;" MMKV_HEADER "${MMKV_HEADER}")
    file(WRITE "${MMKV_CORE_SOURCE}/MMKV.h" "${MMKV_HEADER}")
    file(READ "${MMKV_CORE_SOURCE}/MMKV.cpp" MMKV_SOURCE)
    string(REPLACE "realRoot.ends_with(MMKV_PATH_SLASH)" "(!realRoot.empty() && realRoot.back() == MMKV_PATH_SLASH[0])" MMKV_SOURCE "${MMKV_SOURCE}")
    file(WRITE "${MMKV_CORE_SOURCE}/MMKV.cpp" "${MMKV_SOURCE}")
    file(READ "${MMKV_CORE_SOURCE}/MMKV_IO.cpp" MMKV_IO_SOURCE)
    string(REPLACE "T& container, std::string_view key" "T& container, const std::string &key" MMKV_IO_SOURCE "${MMKV_IO_SOURCE}")
    file(WRITE "${MMKV_CORE_SOURCE}/MMKV_IO.cpp" "${MMKV_IO_SOURCE}")
    if(MINGW)
        file(READ "${MMKV_CORE_SOURCE}/MemoryFile_Win32.cpp" MMKV_WINDOWS_SOURCE)
        string(REPLACE "    __try" "    #undef __try\n    __try" MMKV_WINDOWS_SOURCE "${MMKV_WINDOWS_SOURCE}")
        file(WRITE "${MMKV_CORE_SOURCE}/MemoryFile_Win32.cpp" "${MMKV_WINDOWS_SOURCE}")
    endif()
endif()
add_subdirectory("${MMKV_CORE_SOURCE}" "${CMAKE_CURRENT_BINARY_DIR}/mmkv-core-build")
if(RACHEL_NATIVE_TARGET)
    set_target_properties(core PROPERTIES CXX_STANDARD 17)
    if(MINGW)
        target_compile_options(core PRIVATE -fdelayed-template-parsing -fms-extensions)
    endif()
endif()

add_library(rachel_kv STATIC rachel_kv.cpp)
if(RACHEL_NATIVE_TARGET)
    set_target_properties(rachel_kv PROPERTIES CXX_STANDARD 17)
else()
    set_target_properties(rachel_kv PROPERTIES CXX_STANDARD 20)
endif()
target_include_directories(rachel_kv PRIVATE "${MMKV_CORE_SOURCE}")
target_link_libraries(rachel_kv PRIVATE core)
if(APPLE)
    target_compile_definitions(rachel_kv PRIVATE FORCE_POSIX)
endif()
if(WIN32)
    target_compile_definitions(rachel_kv PRIVATE UNICODE _UNICODE)
endif()
if(MINGW)
    target_compile_options(rachel_kv PRIVATE -fdelayed-template-parsing)
endif()
